%% The header photo. \cvphoto declares it -- the header draws it beside the name later.
%%   \cvphoto                          placeholder photo avatar, default diameter and inset
%%   \cvphoto{yourphoto}               custom image, cropped to a circle
%%   \cvphoto[4cm]                     Diameter setting

%% Follow \cvphoto with a blank line, or it takes a following {...} as its filename -> error.
\cvphoto[4cm]{photo.jpg}

%% Personal information in the resume header. Not all fields are required.
%% A field prints its shown prefix ahead of the value to be self-explanatory
%% (i.e., parser-friendly): \github{your_id} sets `github.com/your_id'
%%
%%   \CvInfoField[link scheme][shown prefix]{fieldname}{symbol}
%%      \CvInfoField[https://][gitlab.com/]{gitlab}{\faGitlab}
%%      \gitlab{your_id}            -> gitlab.com/your_id
%%      \gitlab[noprefix]{your_id}  -> your_id, with the same link
%%   \CvInfoField* takes the full hyperlink as a second argument
%%      \CvInfoField*{mastodon}{\faMastodon}
%%      \mastodon{@username@instance}{https://instance.url/@username}
%%   \cvfieldicon{github}{\faGithubAlt} restyles a field's icon, built-ins too
%% _ & $ ~ ^ may be written literally in a value; only # needs \#.
%% A "$" cannot appear in a link target at all. Escaped "\$" is dropped.
%%
%% Fields predefined by the class:
%% - \email, \phone, \mailaddress, \location, \homepage, \linkedin, \github, \orcid

  \homepage{adriancastellanos.dev}
  \linkedin{adrian-castellanos}
  \github{adrian-castellanos-swe}
  %\orcid{0000-0000-0000-0000}
}
